三、核心实现：Promise 与 async/await

每个"步骤"是一个用 setTimeout 模拟耗时的 Promise（yl_task）。串行步骤用 then 链或 await 依次等待；并行步骤（和面、拌馅）用 Promise.all 同时发起，等两个都完成后才进入下一步。async/await 是 Promise 的语法糖，用类似同步的写法写异步流程。

写法一：Promise 链式调用
yl_task('买原料', 1200, 'yl_stepBuy')
    .then(function () {
        return yl_task('取原料', 900, 'yl_stepPick');   // 买完再取，串行
    })
    .then(function () {
        // Promise.all：和面、拌馅 同时开始，两个都完成才继续
        return Promise.all([
            yl_task('和面', 1800, 'yl_stepDough'),
            yl_task('拌馅', 1400, 'yl_stepFill')
        ]);
    })
    .then(function () {
        return yl_task('包饺子', 1600, 'yl_stepWrap');
    })
    .then(function () {
        return yl_task('吃饺子', 1000, 'yl_stepEat');   // 包完饺子吃饺子
    });
写法二：async/await
async function yl_makeDumplings() {
    await yl_task('买原料', 1200, 'yl_stepBuy');          // 先买原料
    await yl_task('取原料', 900, 'yl_stepPick');          // 后取原料（串行）
    await Promise.all([                                  // 和面、拌馅 同时进行
        yl_task('和面', 1800, 'yl_stepDough'),
        yl_task('拌馅', 1400, 'yl_stepFill')
    ]);                                                  // 两个都完成才继续
    await yl_task('包饺子', 1600, 'yl_stepWrap');
    await yl_task('吃饺子', 1000, 'yl_stepEat');          // 包完饺子吃饺子
}
步骤工厂：每个任务返回一个 Promise
function yl_task(yl_name, yl_ms, yl_stepId) {
    return new Promise(function (yl_resolve) {
        yl_setStep(yl_stepId, 'run');
        yl_log('开始' + yl_name + '…');
        setTimeout(function () {      // 用定时器模拟耗时
            yl_setStep(yl_stepId, 'done');
            yl_log(yl_name + '完成');
            yl_resolve(yl_name);
        }, yl_ms);
    });
}
