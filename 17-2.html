核心实现：fetch + 原生代码

本页完全使用原生代码实现：网络请求使用浏览器内置的 fetch() API（返回 Promise，支持 then 链式调用），数据渲染使用原生 DOM 操作（createElement / textContent / appendChild），页面样式全部内联，未使用 jQuery、Axios 等任何第三方库或 CDN。核心请求代码如下：

// 1. 组装请求地址（原生 URLSearchParams）
var yl_url = 'https://api.open-meteo.com/v1/forecast?' + yl_params.toString();

// 2. fetch 发起请求，返回 Promise
fetch(yl_url)
    .then(function (yl_resp) {
        if (!yl_resp.ok) {
            throw new Error('HTTP ' + yl_resp.status);
        }
        return yl_resp.json(); // 把响应体解析成 JSON 对象
    })
    .then(function (yl_data) {
        yl_renderNow(yl_data.current);     // 原生 DOM 操作渲染实况卡片
        yl_renderChart(yl_data.hourly);    // 渲染今日逐时气温柱状图
        yl_renderForecast(yl_data.daily);  // 渲染未来 5 天预报
    })
    .catch(function (yl_err) {
        yl_setStatus('err', '获取失败：' + yl_err.message); // 捕获网络/解析错误
    });
