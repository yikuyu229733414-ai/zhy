三、核心实现：ECharts 使用要点

ECharts 库已按课程要求下载到本地 cdn/echarts/ 目录（约 1MB 的 echarts.min.js，来自 Apache ECharts 官方发行版），页面通过相对路径 <script src="./cdn/echarts/echarts.min.js"> 引用，完全离线可用。核心步骤：先 echarts.init(dom) 初始化实例，再用 setOption(option) 传入配置（标题、提示框、图例、坐标轴、系列），窗口缩放时调用 resize() 自适应。

<!-- 1. 引用本地 ECharts（不使用远程 CDN） -->
<script src="./cdn/echarts/echarts.min.js"></script>

// 2. 初始化实例并设置配置项
var yl_lineChart = echarts.init(document.getElementById('yl_lineChart'));
var yl_lineOption = {
    title:   { text: '近 7 天网站访问量趋势' },
    tooltip: { trigger: 'axis' },                 // 坐标轴提示框
    legend:  { data: ['PV 浏览量', 'UV 访客数'] },
    xAxis:   { type: 'category', data: ['周一', …, '周日'] },
    yAxis:   { type: 'value' },
    series: [
        { name: 'PV 浏览量', type: 'line', smooth: true, data: [120, 200, …] },
        { name: 'UV 访客数', type: 'line', smooth: true, data: [60, 100, …] }
    ]
};
yl_lineChart.setOption(yl_lineOption);

// 3. 窗口缩放自适应
window.addEventListener('resize', function () {
    yl_lineChart.resize();
    yl_pieChart.resize();
});
